// ===== src/Hello.rs =====
pub struct Hello {

    pub Name: String,
}
impl Hello {

    pub fn new() -> Self {
        Self {
            Name: "Ada",
        }
    }
    pub fn Greet(&mut self) {
        println!("{}", ("Hello, " + self.Name));
    }
    pub fn on_start(&mut self) {
        self.Greet();
    }
}
