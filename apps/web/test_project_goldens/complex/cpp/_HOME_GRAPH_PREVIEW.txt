#include <iostream>
enum class Mode {
    Idle,
    Run
};
class Counter {
public:
    float Total = 0;
    float Limit = 3;
    bool Ready = true;
    Mode CurrentMode = Mode::Run;
    float Add(float n);
    void on_start() {
        if (Ready) {
            std::cout << "go" << std::endl;
            for (int _vvs_i_cx_for_limit = 1; _vvs_i_cx_for_limit <= Limit; _vvs_i_cx_for_limit++) {
                Add(1);
                std::cout << std::to_string(Total) << std::endl;
            }
            switch (CurrentMode) {
              case Mode::Idle:
                std::cout << "Idle" << std::endl;
                    break;
              case Mode::Run:
                std::cout << "Run" << std::endl;
                    break;
            }
        } else {
            std::cout << "skip" << std::endl;
        }
    }
};
float Counter::Add(float n) {
    Total = (Total + n);
    return Total;
}
