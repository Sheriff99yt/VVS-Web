// ===== src/Counter.go =====
import "fmt"
type Mode int
const (
	Mode_Idle Mode = iota
	Mode_Run
)
type Counter struct {
	var Total float64 = 0
	var Limit float64 = 3
	var Ready bool = true
	var CurrentMode Mode = Mode_Run
func Add(n) data_number {
	self.Total = (self.Total + n)
	return self.Total
}
func (self *Counter) on_start() {
	if self.Ready {
		fmt.Println("go")
		for _vvs_i_cx_for_limit := 1; _vvs_i_cx_for_limit <= self.Limit; _vvs_i_cx_for_limit++ {
			self.Add(1)
			fmt.Println(fmt.Sprintf("%v", self.Total))
		}
		switch self.CurrentMode {
		case Mode_Idle:
			fmt.Println("Idle")
		case Mode_Run:
			fmt.Println("Run")
		}
	} else {
		fmt.Println("skip")
	}
}
}
