class SensorStatus(Enum):
    OK = 1
    WARN = 2
    FAIL = 3
class Sensor(Machine):
    Readings = []
    Status = SensorStatus.OK
    Host = None
    Tags = {}
    async def Sample(self):
        self.Readings.append(self.Power)
        print("Sampled")
    def Report(self):
        print("--- Readings ---")
        for val in self.Readings:
            print(str(val))
    def on_tick(self):
        print("Tick")
    def on_start(self):
        match self.Status:
            case SensorStatus.OK:
                self.Sample()
                self.Report()
            case SensorStatus.WARN:
                print("Warn")
                self.Report()
            case SensorStatus.FAIL:
                print("Fail")
                self.Report()
