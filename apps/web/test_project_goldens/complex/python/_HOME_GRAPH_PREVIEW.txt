from enum import Enum
class Mode(Enum):
    Idle = 1
    Run = 2
class Counter:
    Total = 0
    Limit = 3
    Ready = True
    CurrentMode = Mode.Run
    def Add(self, n):
        self.Total = (self.Total + n)
        return self.Total
    def on_start(self):
        if self.Ready:
            print("go")
            for _vvs_i_cx_for_limit in range(1, self.Limit + 1):
                self.Add(1)
                print(str(self.Total))
            match self.CurrentMode:
                case Mode.Idle:
                    print("Idle")
                case Mode.Run:
                    print("Run")
        else:
            print("skip")
