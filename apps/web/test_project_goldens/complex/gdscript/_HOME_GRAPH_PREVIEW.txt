enum Mode { Idle, Run }
class_name Counter
    var Total = 0
    var Limit = 3
    var Ready = true
    var CurrentMode = Mode.Run
    func Add(n) -> void:
        self.Total = (self.Total + n)
        return self.Total
    func on_start() -> void:
        if self.Ready:
            print("go")
            for _vvs_i_cx_for_limit in range(1, self.Limit + 1):
                self.Add(1)
                print(str(self.Total))
            var _vvs_sel = self.CurrentMode
            if _vvs_sel == Mode.Idle:
                print("Idle")
            elif _vvs_sel == Mode.Run:
                print("Run")
        else:
            print("skip")
