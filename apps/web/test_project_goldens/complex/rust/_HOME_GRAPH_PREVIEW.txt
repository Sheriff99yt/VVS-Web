// ===== src/Counter.rs =====
pub enum Mode {
    Idle,
    Run
}
pub struct Counter {

    pub Total: f32,
    pub Limit: f32,
    pub Ready: bool,
    pub CurrentMode: Mode,
}
impl Counter {

    pub fn new() -> Self {
        Self {
            Total: 0,
            Limit: 3,
            Ready: true,
            CurrentMode: Mode::Run,
        }
    }
    pub fn Add(&mut self, n: f32) {
        self.Total = (self.Total + n);
        return self.Total;
    }
    pub fn on_start(&mut self) {
        if self.Ready {
            println!("{}", "go");
            for _vvs_i_cx_for_limit in 1..=self.Limit {
                self.Add(1);
                println!("{}", self.Total.to_string());
            }
            match self.CurrentMode {
                Mode::Idle => {
                    println!("{}", "Idle");
                }
                Mode::Run => {
                    println!("{}", "Run");
                }
            }
        } else {
            println!("{}", "skip");
        }
    }
}
