            Ready: false,
        }
    }
    pub fn Boot(&mut self) {
        self.Ready = true;
        println!("{}", "Booted");
        Sensor::new().on_tick();
    }
    fn Diagnose(&mut self) {
        println!("{}", "Diagnose");
        return;
    }
    pub fn Shutdown(&mut self) {
        self.Ready = false;
        println!("{}", "Shutdown");
    }
    pub fn Calculate(&mut self, a: f32, b: f32) {
        println!("{}", "Calculating");
        return;
    }
    pub fn on_pulse(&mut self) {
        println!("{}", ("Power=" + self.Power.to_string()));
    }
    pub fn on_start(&mut self) {
        print!("{}", "Operator name?");
        std::io::Write::flush(&mut std::io::stdout()).unwrap();

pub enum SensorStatus {
    OK,
    WARN,
    FAIL
}
pub struct Sensor {
    base: Machine,

    pub Readings: Vec<f32>,
    pub Status: SensorStatus,
    pub Host: Machine,
    pub Tags: HashMap<String, String>,
}
impl Sensor {

    pub fn new() -> Self {
        Self {
            base: Machine::new(),
            Readings: Vec::new(),
            Status: SensorStatus::OK,
            Host: Machine::new(),
            Tags: HashMap::new(),
        }
    }
    pub fn Sample(&mut self) {
        self.Readings.push(self.base.Power);
        println!("{}", "Sampled");
    }
    pub fn Report(&mut self) {
        println!("{}", "--- Readings ---");
        for val in self.Readings.iter() {
            println!("{}", val.to_string());
        }
    }
    pub fn on_tick(&mut self) {
        println!("{}", "Tick");
    }
    pub fn on_start(&mut self) {
        match self.Status {
            SensorStatus::OK => {
                self.Sample();
                self.Report();
            }
            SensorStatus::WARN => {
                println!("{}", "Warn");
                self.Report();
            }
            SensorStatus::FAIL => {
                println!("{}", "Fail");
                self.Report();
            }
        }
    }
}
