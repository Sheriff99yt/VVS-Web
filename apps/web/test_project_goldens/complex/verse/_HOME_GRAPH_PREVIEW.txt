<public>Machine := class:
    var Power : float = 0
    var Serial<public> : float = 0
    var MaxPower<public> : float = 100
    var Ready<public> : logic = false
    Boot<public>() : void =
        set Ready = true
        Print("Booted")
        Sensor.on_tick()
    Diagnose() : void =
        Print("Diagnose")
        return
    Shutdown<public>() : void =
        set Ready = false
        Print("Shutdown")
    Calculate<public>(a : float, b : float) : void =
        Print("Calculating")
        return (a + b)
    on_pulse() : void =
        Print(("Power=" + ToString(Power)))
    on_start() : void =
        Print("Operator name?")
        # (x) Get User Input
        var _vvs_input_lab_get_input : string = ""
        Print(_vvs_input_lab_get_input)
        Boot()
        Print(ToString(Calculate(Power, MaxPower)))
        if (Ready?):
            on_pulse()
        else:
            Print("Not ready")
SensorStatus := enum:
    OK
    WARN
    FAIL
<public>Sensor(Machine) := class:
    var Readings<public> : []float = array{}
    var Status<public> : SensorStatus = SensorStatus.OK
    var Host<public> : Machine = Machine{}
    var Tags<public> : [string]string = map{}
    Sample<public>() : void =
        set Readings += array{Power}
        Print("Sampled")
    Report<public><override>() : void =
        Print("--- Readings ---")
        for (val : Readings):
            Print(ToString(val))
    on_tick() : void =
        Print("Tick")
    on_start<override>() : void =
        # switch (Status)
        if (Status = SensorStatus.OK):
            Sample()
            Report()
        if (Status = SensorStatus.WARN):
            Print("Warn")
            Report()
        if (Status = SensorStatus.FAIL):
            Print("Fail")
            Report()
