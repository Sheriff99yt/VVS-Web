class MainClass {
public:
    std::string GlobalGreeting = "World";
    float ProcessData(float a, float b);
    std::string ProcessData(std::string a, std::string b);
    void on_ondatareceived(std::string payload) {
        ProcessData(payload, payload);
        std::cout << "/* Call ProcessData.return_val */" << std::endl;
    }
    void on_start() {
        on_ondatareceived(("Hello from Global: " + GlobalGreeting));
    }
};
float MainClass::ProcessData(float a, float b) {
    return (a + b);
}
std::string MainClass::ProcessData(std::string a, std::string b) {
    return (a + b);
    // Concat Strings
}
