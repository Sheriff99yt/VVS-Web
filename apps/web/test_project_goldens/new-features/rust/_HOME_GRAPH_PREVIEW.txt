// ===== src/NewFeaturesLab.rs =====
pub struct MainClass {

    pub GlobalGreeting: String,
}
impl MainClass {

    pub fn new() -> Self {
        Self {
            GlobalGreeting: "World",
        }
    }
    pub fn ProcessData(&mut self, a: f32, b: f32) {
        return (a + b);
    }
    // (x) Implement ProcessData
    pub fn on_ondatareceived(&mut self, payload: String) {
        self.ProcessData(payload, payload);
        println!("{}", "/* Call ProcessData.return_val */");
    }
    pub fn on_start(&mut self) {
        self.on_ondatareceived(("Hello from Global: " + self.GlobalGreeting));
    }
}
