<public>MainClass := class:
    var GlobalGreeting<public> : string = "World"
    ProcessData<public>(a : float, b : float) : void =
        return (a + b)
    ProcessData<public>(a : string, b : string) : void =
        return (a + b)
    on_ondatareceived(payload : string) : void =
        Print(ProcessData(payload, payload))
    on_start() : void =
        on_ondatareceived(("Hello from Global: " + GlobalGreeting))
