// ===== src/InheritanceLab.go =====
import "time"
import "fmt"
type Parent struct {
func Speak() void {
	fmt.Println("parent")
}
}
type Child struct {
func Speak() void {
	self.Parent.Speak()
	fmt.Println("child")
}
func (self *Child) on_start() {
	time.Sleep(time.Duration(1 * float64(time.Second)))
	self.Speak()
	fmt.Println("done")
}
}
