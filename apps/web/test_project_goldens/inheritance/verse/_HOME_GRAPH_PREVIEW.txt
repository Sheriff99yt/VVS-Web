<public>Parent := class:
    Speak<public>() : void =
        Print("parent")
<public>Child(Parent) := class:
    Speak<public><override>() : void =
        (super:)Speak()
        Print("child")
    on_start() : void =
        # wait 1s
        Speak()
        Print("done")
