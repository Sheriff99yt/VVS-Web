class_name Parent
    # (x) Declare Parent
    func _init():
        pass
    # (x) Declare Speak
    func Speak() -> void:
        print("parent")
class_name Child
extends Parent
    # (x) Declare Speak
    func Speak() -> void:
        super.Speak()
        print("child")
    func on_start() -> void:
        await get_tree().create_timer(1).timeout
        self.Speak()
        print("done")
