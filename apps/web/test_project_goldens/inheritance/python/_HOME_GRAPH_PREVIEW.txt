import asyncio
class Parent:
    def __init__(self):
        pass
    def Speak(self):
        print("parent")
class Child(Parent):
    def Speak(self):
        super().Speak()
        print("child")
    async def on_start(self):
        await asyncio.sleep(1)
        self.Speak()
        print("done")
