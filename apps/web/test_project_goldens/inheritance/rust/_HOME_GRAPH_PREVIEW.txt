// ===== src/InheritanceLab.rs =====
pub struct Parent {

}
impl Parent {

    pub fn new() -> Self {
        Self {
        }
    }
    pub fn Speak(&mut self) {
        println!("{}", "parent");
    }
}

pub struct Child {
    base: Parent,

}
impl Child {

    pub fn new() -> Self {
        Self {
            base: Parent::new(),
        }
    }
    pub fn Speak(&mut self) {
        self.base.Speak();
        println!("{}", "child");
    }
    pub fn on_start(&mut self) {
        std::thread::sleep(std::time::Duration::from_secs_f64(1));
        self.Speak();
        println!("{}", "done");
    }
}
